def stamp():
 now=datetime.now(timezone.utc)
 return {'fecha_utc':now.isoformat(),'fecha_caracas':now.astimezone(ZoneInfo('America/Caracas')).isoformat()}

def readrow():return trabajo.read_task(BASE,IDENT)

def meta(pid):
 try:
  p=Path('/proc',str(pid));fields=(p/'stat').read_text().rsplit(')',1)[1].split();start=hs.process_start(pid)
  if not start:return None
  return {'pid':pid,'start':start,'ppid':int(fields[1]),'pgrp':int(fields[2]),'sid_os':int(fields[3]),'estado_proc':fields[0],'exe':str((p/'exe').resolve()),'cwd':str((p/'cwd').resolve()),'uid':p.stat().st_uid}
 except (FileNotFoundError,ProcessLookupError):return None

def descendants(pid):
 parents={}
 for p in Path('/proc').iterdir():
  if not p.name.isdigit():continue
  try:parents[int(p.name)]=int((p/'stat').read_text().rsplit(')',1)[1].split()[1])
  except (FileNotFoundError,ProcessLookupError,PermissionError,ValueError):pass
 found={pid}
 while True:
  added={child for child,parent in parents.items() if parent in found}-found
  if not added:break
  found.update(added)
 return [m for child in sorted(found-{pid}) if (m:=meta(child))]

def git_snapshot(session):
 wt=Path(session['worktree']);route=wt/'_investigacion-onia/base-operativa'/CFG['route']
 def git(*a):return hs.git(wt,'--no-optional-locks',*a)
 files={name:hashlib.sha256((route/name).read_bytes()).hexdigest() for name in ('checkpoint.json','nota.txt')}
 assert (route/'checkpoint.json').read_bytes()==CFG['checkpoint'].encode()
 assert (route/'nota.txt').read_bytes()==CFG['note'].encode()
 index=Path(git('rev-parse','--git-path','index'))
 return {'files_sha256':files,'index_sha256':hashlib.sha256(index.read_bytes()).hexdigest(),'index_entries':git('ls-files','--stage','--','_investigacion-onia/base-operativa/'+CFG['route']),'status':git('status','--porcelain','--untracked-files=all'),'head':git('rev-parse','HEAD'),'commit_count':int(git('rev-list','--count',session['commit_base']+'..HEAD')),'remote':git('ls-remote','origin','refs/heads/'+session['rama'])}
